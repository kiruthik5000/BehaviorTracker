<h1 class="font-headline-lg text-headline-lg text-primary mb-md">Two Sum</h1>
<div class="prose prose-invert max-w-none text-on-surface-variant text-[13px] leading-relaxed">
<p class="mb-4">Given an array of integers <code class="bg-surface-container-high px-1 rounded text-primary">nums</code> and an integer <code class="bg-surface-container-high px-1 rounded text-primary">target</code>, return indices of the two numbers such that they add up to target.</p>
<p class="mb-4">You may assume that each input would have <strong>exactly one solution</strong>, and you may not use the same element twice.</p>
<p class="mb-md">You can return the answer in any order.</p>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Example 1:</h3>
<div class="bg-surface-container-lowest p-md rounded border border-outline-variant font-code-md">
<p><span class="text-on-surface-variant">Input:</span> nums = [2,7,11,15], target = 9</p>
<p><span class="text-on-surface-variant">Output:</span> [0,1]</p>
<p><span class="text-on-surface-variant">Explanation:</span> Because nums[0] + nums[1] == 9, we return [0, 1].</p>
</div>
</div>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Example 2:</h3>
<div class="bg-surface-container-lowest p-md rounded border border-outline-variant font-code-md">
<p><span class="text-on-surface-variant">Input:</span> nums = [3,2,4], target = 6</p>
<p><span class="text-on-surface-variant">Output:</span> [1,2]</p>
</div>
</div>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Constraints:</h3>
<ul class="list-disc pl-5 space-y-1">
<li>2 &lt;= nums.length &lt;= 10<sup>4</sup></li>
<li>-10<sup>9</sup> &lt;= nums[i] &lt;= 10<sup>9</sup></li>
<li>-10<sup>9</sup> &lt;= target &lt;= 10<sup>9</sup></li>
<li class="text-tertiary">Only one valid answer exists.</li>
</ul>
</div>
<div class="p-md bg-surface-container-high rounded-lg border border-outline-variant/30 italic">
<span class="text-tertiary font-bold not-italic mr-2">Follow-up:</span> Can you come up with an algorithm that is less than O(n<sup>2</sup>) time complexity?
                        </div>
</div>

<span class="code-syntax-comment">/**
 * @param {number[]} nums
 * @param {number} target
 * @return {number[]}
 */</span>
<span class="code-syntax-keyword">const</span> <span class="code-syntax-function">twoSum</span> = <span class="code-syntax-keyword">function</span>(nums, target) {
    <span class="code-syntax-keyword">const</span> map = <span class="code-syntax-keyword">new</span> <span class="code-syntax-function">Map</span>();
    <span class="code-syntax-keyword">for</span> (<span class="code-syntax-keyword">let</span> i = <span class="number">0</span>; i &lt; nums.length; i++) {
        <span class="code-syntax-keyword">const</span> complement = target - nums[i];
        <span class="code-syntax-keyword">if</span> (map.<span class="code-syntax-function">has</span>(complement)) {
            <span class="code-syntax-keyword">return</span> [map.<span class="code-syntax-function">get</span>(complement), i];
        }
        map.<span class="code-syntax-function">set</span>(nums[i], i);
    }
};</div>

<h1 class="font-headline-lg text-headline-lg text-primary mb-md">Two Sum</h1>
<div class="prose prose-invert max-w-none text-on-surface-variant text-[13px] leading-relaxed">
<p class="mb-4">Given an array of integers <code class="bg-surface-container-high px-1 rounded text-primary">nums</code> and an integer <code class="bg-surface-container-high px-1 rounded text-primary">target</code>, return indices of the two numbers such that they add up to target.</p>
<p class="mb-4">You may assume that each input would have <strong>exactly one solution</strong>, and you may not use the same element twice.</p>
<p class="mb-md">You can return the answer in any order.</p>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Example 1:</h3>
<div class="bg-surface-container-lowest p-md rounded border border-outline-variant font-code-md">
<p><span class="text-on-surface-variant">Input:</span> nums = [2,7,11,15], target = 9</p>
<p><span class="text-on-surface-variant">Output:</span> [0,1]</p>
<p><span class="text-on-surface-variant">Explanation:</span> Because nums[0] + nums[1] == 9, we return [0, 1].</p>
</div>
</div>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Example 2:</h3>
<div class="bg-surface-container-lowest p-md rounded border border-outline-variant font-code-md">
<p><span class="text-on-surface-variant">Input:</span> nums = [3,2,4], target = 6</p>
<p><span class="text-on-surface-variant">Output:</span> [1,2]</p>
</div>
</div>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Constraints:</h3>
<ul class="list-disc pl-5 space-y-1">
<li>2 &lt;= nums.length &lt;= 10<sup>4</sup></li>
<li>-10<sup>9</sup> &lt;= nums[i] &lt;= 10<sup>9</sup></li>
<li>-10<sup>9</sup> &lt;= target &lt;= 10<sup>9</sup></li>
<li class="text-tertiary">Only one valid answer exists.</li>
</ul>
</div>
<div class="p-md bg-surface-container-high rounded-lg border border-outline-variant/30 italic">
<span class="text-tertiary font-bold not-italic mr-2">Follow-up:</span> Can you come up with an algorithm that is less than O(n<sup>2</sup>) time complexity?
                    </div>

<span class="code-syntax-keyword">const</span> <span class="code-syntax-function">twoSum</span> = <span class="code-syntax-keyword">function</span>(nums, target) {
    <span class="code-syntax-keyword">const</span> map = <span class="code-syntax-keyword">new</span> <span class="code-syntax-function">Map</span>();

<h1 class="font-headline-lg text-headline-lg text-primary mb-md">Two Sum</h1>
<div class="prose prose-invert max-w-none text-on-surface-variant text-[13px] leading-relaxed">
<p class="mb-4">Given an array of integers <code class="bg-surface-container-high px-1 rounded text-primary">nums</code> and an integer <code class="bg-surface-container-high px-1 rounded text-primary">target</code>, return indices of the two numbers such that they add up to target.</p>
<p class="mb-4">You may assume that each input would have <strong>exactly one solution</strong>, and you may not use the same element twice.</p>
<p class="mb-md">You can return the answer in any order.</p>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Example 1:</h3>
<div class="bg-surface-container-lowest p-md rounded border border-outline-variant font-code-md">
<p><span class="text-on-surface-variant">Input:</span> nums = [2,7,11,15], target = 9</p>
<p><span class="text-on-surface-variant">Output:</span> [0,1]</p>
<p><span class="text-on-surface-variant">Explanation:</span> Because nums[0] + nums[1] == 9, we return [0, 1].</p>
</div>
</div>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Example 2:</h3>
<div class="bg-surface-container-lowest p-md rounded border border-outline-variant font-code-md">
<p><span class="text-on-surface-variant">Input:</span> nums = [3,2,4], target = 6</p>
<p><span class="text-on-surface-variant">Output:</span> [1,2]</p>
</div>
</div>
<div class="mb-6">
<h3 class="text-on-surface font-bold mb-2">Constraints:</h3>
<ul class="list-disc pl-5 space-y-1">
<li>2 &lt;= nums.length &lt;= 10<sup>4</sup></li>
<li>-10<sup>9</sup> &lt;= nums[i] &lt;= 10<sup>9</sup></li>
<li>-10<sup>9</sup> &lt;= target &lt;= 10<sup>9</sup></li>
<li class="text-tertiary">Only one valid answer exists.</li>
</ul>
</div>
<div class="p-md bg-surface-container-high rounded-lg border border-outline-variant/30 italic">
<span class="text-tertiary font-bold not-italic mr-2">Follow-up:</span> Can you come up with an algorithm that is less than O(n<sup>2</sup>) time complexity?

<span class="code-syntax-keyword">const</span> <span class="code-syntax-function">twoSum</span> = <span class="code-syntax-keyword">function</span>(nums, target) {
    <span class="code-syntax-keyword">const</span> map = <span class="code-syntax-keyword">new</span> <span class="code-syntax-function">Map</span>();
    <span class="code-syntax-keyword">for</span> (<span class="code-syntax-keyword">let</span> i = <span class="number">0</span>; i &lt; nums.length; i++) {
